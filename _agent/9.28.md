### 手撕题
一排灯笼，每个灯笼要么是亮着的，要么是坏掉的。要求摘掉一个坏掉的灯笼，摘掉一个坏灯笼之后，最多能有多少个亮着的灯笼连续排在一起？输出这个最大数量。

**【思路】记录坏掉的下标：后一个-前一个，左+右**

```cpp
#include<bits/stdc++.h>
using namespace std;
int main(){
    string s;
    cin>>s;
    vector<int> zeros;
    for(int i=0;i<s.size();i++){
        if(s[i]=='0') zeros.push_back(i);
    }

    int res=0;
    for(int i=0;i<zeros.size();i++){
        //		左
        int left_count=0;
        if(i==0) left_count=zeros[i]-1-0+1;
        else left_count=zeros[i]-1-(zeros[i-1]+1)+1;
        //		右 
        int right_count=0;	
        if(i==zeros.size()-1) right_count=s.size()-1-(zeros[i]+1)+1;
        else right_count=zeros[i+1]-1-(zeros[i]+1)+1;
        res=max(res,left_count+right_count);
    }
    cout<<res-1<<endl;
    return 0;
}
```

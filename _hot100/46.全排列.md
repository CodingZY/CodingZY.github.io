## 📌 题目

> 给定一个不含重复数字的数组 nums ，返回其 所有可能的全排列 。你可以 按任意顺序 返回答案。

## 💡 思路
- res,state,nums,selected


## 💻 代码

```cpp
void dfs(vector<vector<int>> &res,vector<int> &state,vector<int>& nums,vector<bool> &selected){
    if(state.size()==nums.size()){
        res.push_back(state);
        return;
    }
    for(int i=0;i<nums.size();i++){
        if(selected[i]==false){
            state.push_back(nums[i]);
            selected[i]=true;
            dfs(res,state,nums,selected);
            selected[i]=false;
            state.pop_back();
        }
    }
}
vector<vector<int>> permute(vector<int>& nums) {
    int n=nums.size();
    vector<vector<int>> res;
    vector<int> state;
    vector<bool> selected(n,false);
    dfs(res,state,nums,selected);
    return res;
}
```

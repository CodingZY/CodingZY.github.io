## 📌 题目

> 给你一个整数数组 nums ，数组中的元素 互不相同 。返回该数组所有可能的子集（幂集）。解集 不能 包含重复的子集。你可以按 任意顺序 返回解集。

## 💡 思路
- 递归出口+循环写法
- res,state,nums,index


## 💻 代码

```cpp
void dfs(vector<vector<int>> &res,vector<int> &state,vector<int> &nums,int index){
    if(index==nums.size()){
        res.push_back(state);
        return;
    }
    state.push_back(nums[index]);
    dfs(res,state,nums,index+1);
    state.pop_back();
    dfs(res,state,nums,index+1);
}
vector<vector<int>> subsets(vector<int>& nums) {
    int n=nums.size();
    vector<vector<int>> res;
    vector<int> state;
    dfs(res,state,nums,0);
    return res;
}

方法二：
void dfs(vector<vector<int>>& res, vector<int>& state, vector<int>& nums, int start) {
    // 每一个节点都是一个合法的子集，直接存入 res
    res.push_back(state);

    for (int i = start; i < nums.size(); i++) {
        state.push_back(nums[i]);  // 选择元素
        dfs(res, state, nums, i + 1); // 递归下一个位置
        state.pop_back();          // 回溯
    }
}

vector<vector<int>> subsets(vector<int>& nums) {
    vector<vector<int>> res;
    vector<int> state;
    dfs(res, state, nums, 0);
    return res;
}
```

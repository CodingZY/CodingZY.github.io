题目：
给定一个字符串 s ，请你找出其中不含有重复字符的 最长 子串 的长度

思路：
子串是连续的->滑动窗口
unordered_map<char,int>=?0

代码：
int lengthOfLongestSubstring(string s) {
    int res=0;
    int left=0,right=0;
    unordered_map<char,int> map;
    for(;right<s.size();right++){
        while(left<right&&map[s[right]]!=0){
            map[s[left]]--;
            left++;
        }
        map[s[right]]++;
        res=max(right-left+1,res);
    }
    return res;
}
